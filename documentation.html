Quick Start

5 minute setup
bash 
# Clone the starter template
git clone https://github.com/barrios-a2i/orchestrator-starter.git
cd orchestrator-starter

# Install dependencies
pip install -r requirements.txt

# Configure your environment
cp .env.example .env
# Edit .env with your API keys

# Start the orchestration server
python -m barrios_orchestrator serve

# In another terminal, deploy your first agent
python -m barrios_orchestrator deploy agents/example_agent.py

# View the dashboard
open http://localhost:8080/dashboard
